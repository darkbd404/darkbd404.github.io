import smtplib from email.mime.text import MIMEText from email.mime.multipart import MIMEMultipart def send_html_email(receiver_email, subject, html_body): """ HTML ফরম্যাটে ইমেইল পাঠান """ sender_email = "your_email@gmail.com" app_password = "xxxx xxxx xxxx xxxx" try: msg = MIMEMultipart('alternative') msg['From'] = sender_email msg['To'] = receiver_email msg['Subject'] = subject # HTML body html_part = MIMEText(html_body, 'html') msg.attach(html_part) server = smtplib.SMTP('smtp.gmail.com', 587) server.ehlo() server.starttls() server.ehlo() server.login(sender_email, app_password) server.sendmail(sender_email, receiver_email, msg.as_string()) server.quit() print(f"✅ HTML ইমেইল পাঠানো হয়েছে: {receiver_email}") return True except Exception as e: print(f"❌ Error: {str(e)}") return False # HTML টেমপ্লেট উদাহরণ html_template = """

Hello!

This is an HTML email sent from Python.

Best regards,
Salam

""" send_html_email("receiver@example.com", "HTML Test", html_template)
